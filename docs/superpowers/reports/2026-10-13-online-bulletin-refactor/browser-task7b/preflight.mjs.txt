import {createRequire} from 'node:module';
import {spawn} from 'node:child_process';
import {once} from 'node:events';
import assert from 'node:assert/strict';
const req=createRequire('/Users/rayselfs/Projects/hhc/website/.worktrees/online-bulletin-refactor-20261013/frontend-platform/package.json');
const {chromium}=req('playwright');
const server=spawn(process.execPath,['/tmp/bulletin-task7b-browser/rollback-server.mjs','/tmp/bulletin-web-baseline-20261013'],{cwd:'/Users/rayselfs/Projects/hhc/website/.worktrees/online-bulletin-refactor-20261013/hhc-web-capabilities',stdio:['ignore','pipe','pipe']});
